];


const result = operations.reduce((acc, { date }) => {
    const [year, ...rest] = date.split('-');
    const dayMonth = rest.join('-');
    
    acc[year] = acc[year] ? acc[year].concat(dayMonth).sort() : [dayMonth];
    
    return acc;    
}, {});

console.log(result);

### Правильное решение

```js
const result = [...operations]
  .sort((a, b) => a.date.localeCompare(b.date)) // либо operations.toSorted((a, b) => a.date.localeCompare(b.date))
  .reduce((acc, {date}) => {
    const [year, month, day] = date.split('-');

    acc[year] ??= [];
    acc[year].push(`${month}-${day}`);

    return acc;
  }, {});
```

- Еще один вариант через Object.groupBy, Object.fromEntries

```js
const grouped = Object.groupBy(operations, ({date}) => date.slice(0, 4));

const result = Object.fromEntries(
  Object.entries(grouped).map(([year, operations]) => [
    year,
    operations
      .map(({date}) => date.slice(5))
      .sort(),
  ]),
);

console.log(result);
```
